cmake_minimum_required(VERSION 3.20)
project(show_tools_patches LANGUAGES CXX)
add_library(patch_engine src/patch_engine.cpp)
target_include_directories(patch_engine PUBLIC include)
target_compile_features(patch_engine PUBLIC cxx_std_17)
target_compile_options(patch_engine PRIVATE -Wall -Wextra -Wpedantic -fno-exceptions)
include(CTest)
if(BUILD_TESTING)
  add_executable(patch_engine_tests tests/native_tests.cpp)
  target_link_libraries(patch_engine_tests PRIVATE patch_engine)
  target_compile_options(patch_engine_tests PRIVATE -Wall -Wextra -Wpedantic -fno-exceptions)
  add_test(NAME patch_engine COMMAND patch_engine_tests)
endif()
