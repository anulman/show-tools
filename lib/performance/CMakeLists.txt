cmake_minimum_required(VERSION 3.20)
project(show_tools_performance LANGUAGES CXX)
include(../peripherals/light-math.cmake)
add_library(performance_light_bank src/performance_lights.cpp)
target_include_directories(performance_light_bank PRIVATE src)
target_link_libraries(performance_light_bank PUBLIC peripheral_light_math)
target_compile_features(performance_light_bank PUBLIC cxx_std_17)
target_compile_options(performance_light_bank PRIVATE -Wall -Wextra -Wpedantic -fno-exceptions)
if(NOT TARGET audio_sequence_engine)
  add_subdirectory(../sequencer ${CMAKE_CURRENT_BINARY_DIR}/sequencer)
endif()
add_library(performance_clock
  ../beat-tracker/src/beat_tracker.cpp
  ../beat-tracker/src/beat_clock.cpp
  ../beat-tracker/src/tempo_follower.cpp)
target_include_directories(performance_clock PUBLIC ../beat-tracker/include)
target_compile_features(performance_clock PUBLIC cxx_std_17)
target_compile_definitions(performance_clock PRIVATE BT_MANUAL_ONLY)
target_compile_options(performance_clock PRIVATE -Wall -Wextra -Wpedantic -fno-exceptions)
if(NOT TARGET patch_engine)
  add_subdirectory(../patches ${CMAKE_CURRENT_BINARY_DIR}/patches)
endif()
add_library(performance_engine src/performance_engine.cpp src/compiled_program.cpp)
target_include_directories(performance_engine PUBLIC include)
target_link_libraries(performance_engine PUBLIC performance_clock audio_sequence_engine patch_engine)
target_compile_features(performance_engine PUBLIC cxx_std_17)
target_compile_options(performance_engine PRIVATE -Wall -Wextra -Wpedantic -fno-exceptions)
include(CTest)
if(BUILD_TESTING)
  add_executable(performance_light_bank_tests tests/light_bank_tests.cpp)
  target_include_directories(performance_light_bank_tests PRIVATE src)
  target_link_libraries(performance_light_bank_tests PRIVATE performance_light_bank)
  add_test(NAME performance_light_bank COMMAND performance_light_bank_tests)
  add_executable(compiled_song_host tests/compiled_song_host.cpp)
  target_link_libraries(compiled_song_host PRIVATE performance_engine)
  target_compile_options(compiled_song_host PRIVATE -Wall -Wextra -Wpedantic)
  add_executable(performance_engine_tests tests/native_tests.cpp)
  target_link_libraries(performance_engine_tests PRIVATE performance_engine)
  add_test(NAME performance_engine COMMAND performance_engine_tests)
  add_executable(compiled_program_tests tests/compiled_program_tests.cpp)
  target_link_libraries(compiled_program_tests PRIVATE performance_engine)
  add_test(NAME compiled_program COMMAND compiled_program_tests)
endif()
