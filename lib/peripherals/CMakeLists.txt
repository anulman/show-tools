cmake_minimum_required(VERSION 3.20)
project(show_tools_peripherals LANGUAGES CXX)
add_library(peripheral_controls src/peripheral_controls.cpp)
target_include_directories(peripheral_controls PUBLIC include)
target_compile_features(peripheral_controls PUBLIC cxx_std_17)
target_compile_options(peripheral_controls PRIVATE -Wall -Wextra -Wpedantic)
if(APPLE)
  add_executable(peripheral-probe host/probe.cpp host/macos.cpp)
  target_link_libraries(peripheral-probe PRIVATE peripheral_controls "-framework IOKit" "-framework CoreFoundation")
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  add_executable(peripheral-probe host/probe.cpp host/linux.cpp)
  target_link_libraries(peripheral-probe PRIVATE peripheral_controls)
endif()
if(TARGET peripheral-probe)
  target_compile_options(peripheral-probe PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(peripheral_controls_tests tests/core_cases.cpp)
  target_link_libraries(peripheral_controls_tests PRIVATE peripheral_controls)
  add_test(NAME peripheral_controls COMMAND peripheral_controls_tests)
  if(TARGET peripheral-probe)
    add_test(NAME peripheral_probe_help COMMAND peripheral-probe --help)
    add_test(NAME peripheral_probe_refuse_unconfirmed_output COMMAND peripheral-probe led not-a-device 0 1)
    set_tests_properties(peripheral_probe_refuse_unconfirmed_output PROPERTIES WILL_FAIL TRUE)
  endif()
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../performance/CMakeLists.txt")
    add_subdirectory(../performance ${CMAKE_CURRENT_BINARY_DIR}/performance)
    add_executable(peripheral_performance_fixture tests/performance_fixture.cpp)
    target_link_libraries(peripheral_performance_fixture PRIVATE peripheral_controls performance_engine)
    add_test(NAME peripheral_performance COMMAND peripheral_performance_fixture)
  endif()
endif()
