cmake_minimum_required(VERSION 3.20)
project(show_tools_peripherals LANGUAGES CXX)
include(light-math.cmake)
add_library(peripheral_controls src/peripheral_controls.cpp src/peripheral_light_delivery.cpp)
target_include_directories(peripheral_controls PRIVATE ../performance/include)
target_link_libraries(peripheral_controls PUBLIC peripheral_light_math)
target_include_directories(peripheral_controls PUBLIC include)
target_compile_features(peripheral_controls PUBLIC cxx_std_17)
target_compile_options(peripheral_controls PRIVATE -Wall -Wextra -Wpedantic)
if(APPLE)
  add_executable(peripheral-host host/macos_host.cpp host/control_stream.cpp)
  target_include_directories(peripheral-host PRIVATE ../performance/include)
  target_link_libraries(peripheral-host PRIVATE peripheral_controls "-framework IOKit" "-framework CoreFoundation")
  target_compile_options(peripheral-host PRIVATE -Wall -Wextra -Wpedantic)
  add_executable(peripheral-probe host/probe.cpp host/led_output.cpp host/macos.cpp)
  target_link_libraries(peripheral-probe PRIVATE peripheral_controls "-framework IOKit" "-framework CoreFoundation")
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  add_executable(peripheral-probe host/probe.cpp host/led_output.cpp host/linux.cpp)
  target_link_libraries(peripheral-probe PRIVATE peripheral_controls)
endif()
if(TARGET peripheral-probe)
  target_compile_options(peripheral-probe PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(peripheral_host_stream_tests tests/control_stream.cpp host/control_stream.cpp)
  target_include_directories(peripheral_host_stream_tests PRIVATE host ../performance/include)
  target_link_libraries(peripheral_host_stream_tests PRIVATE peripheral_controls)
  add_test(NAME peripheral_host_stream COMMAND peripheral_host_stream_tests)
  if(TARGET peripheral-host)
    add_test(NAME peripheral_host_help COMMAND peripheral-host --help)
    add_test(NAME peripheral_host_refuse_unconfirmed COMMAND peripheral-host 1 0)
    set_tests_properties(peripheral_host_refuse_unconfirmed PROPERTIES WILL_FAIL TRUE)
    add_test(NAME peripheral_host_refuse_bench_unconfirmed COMMAND peripheral-host --bench-held 0)
    add_test(NAME peripheral_host_refuse_bench_invalid_channel COMMAND peripheral-host --bench-held 96 --confirm-wiring)
    set_tests_properties(peripheral_host_refuse_bench_unconfirmed peripheral_host_refuse_bench_invalid_channel PROPERTIES WILL_FAIL TRUE)
  endif()
  add_executable(peripheral_controls_tests tests/core_cases.cpp tests/delivery_cases.cpp)
  target_include_directories(peripheral_controls_tests PRIVATE ../performance/include)
  target_link_libraries(peripheral_controls_tests PRIVATE peripheral_controls)
  add_test(NAME peripheral_controls COMMAND peripheral_controls_tests)
  add_executable(peripheral_led_output_tests tests/led_output.cpp host/led_output.cpp)
  target_include_directories(peripheral_led_output_tests PRIVATE host)
  target_link_libraries(peripheral_led_output_tests PRIVATE peripheral_controls)
  add_test(NAME peripheral_led_output COMMAND peripheral_led_output_tests)
  if(TARGET peripheral-probe)
    add_test(NAME peripheral_probe_help COMMAND peripheral-probe --help)
    add_test(NAME peripheral_probe_refuse_unconfirmed_output COMMAND peripheral-probe led not-a-device 0 1)
    set_tests_properties(peripheral_probe_refuse_unconfirmed_output PROPERTIES WILL_FAIL TRUE)
    foreach(command led-on led-pulse)
      add_test(NAME peripheral_probe_refuse_${command} COMMAND peripheral-probe ${command} not-a-device 0)
      add_test(NAME peripheral_probe_refuse_invalid_${command} COMMAND peripheral-probe ${command} not-a-device 96 --confirm-wiring)
      set_tests_properties(peripheral_probe_refuse_${command} peripheral_probe_refuse_invalid_${command} PROPERTIES WILL_FAIL TRUE)
    endforeach()
  endif()
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../performance/CMakeLists.txt")
    add_subdirectory(../performance ${CMAKE_CURRENT_BINARY_DIR}/performance)
    add_executable(peripheral_performance_fixture tests/performance_fixture.cpp)
    target_link_libraries(peripheral_performance_fixture PRIVATE peripheral_controls performance_engine)
    add_test(NAME peripheral_performance COMMAND peripheral_performance_fixture)
  endif()
endif()
