cmake_minimum_required(VERSION 3.20)
project(beat_tracker LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(AUBIO REQUIRED IMPORTED_TARGET aubio>=0.4.9)
pkg_check_modules(SNDFILE REQUIRED IMPORTED_TARGET sndfile)
add_library(beat_tracker src/beat_tracker.cpp src/beat_clock.cpp src/aubio_detector.cpp src/tempo_follower.cpp)
target_link_libraries(beat_tracker PRIVATE PkgConfig::AUBIO)
add_executable(beat-tracker src/main.cpp)
target_link_libraries(beat-tracker PRIVATE beat_tracker PkgConfig::SNDFILE)
target_include_directories(beat_tracker PUBLIC include)
target_compile_options(beat_tracker PRIVATE -Wall -Wextra -Wpedantic)
include(CTest)
if(BUILD_TESTING)
  add_executable(clock_tests tests/clock_tests.cpp)
  target_link_libraries(clock_tests PRIVATE beat_tracker)
  add_test(NAME clock COMMAND clock_tests)
  add_executable(follower_tests tests/follower_tests.cpp)
  target_link_libraries(follower_tests PRIVATE beat_tracker)
  add_test(NAME follower COMMAND follower_tests)
  add_executable(detector_tests tests/detector_tests.cpp)
  target_link_libraries(detector_tests PRIVATE beat_tracker)
  add_test(NAME detector COMMAND detector_tests)
endif()
