cmake_minimum_required(VERSION 3.20)
project(show_tools_native_host LANGUAGES CXX)
include(CTest)
add_subdirectory(../performance ${CMAKE_CURRENT_BINARY_DIR}/performance)
add_library(native_audio_pump audio_pump.cpp frame.cpp instrument.cpp)
target_link_libraries(native_audio_pump PUBLIC performance_engine)
target_include_directories(native_audio_pump PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})

target_compile_features(native_audio_pump PUBLIC cxx_std_17)
target_compile_options(native_audio_pump PRIVATE -Wall -Wextra -Wpedantic)
find_package(Threads REQUIRED)
add_subdirectory(../native-alsa ${CMAKE_CURRENT_BINARY_DIR}/alsa)
add_executable(native_fake_host service.cpp)
target_link_libraries(native_fake_host PRIVATE native_audio_pump Threads::Threads)
target_compile_options(native_fake_host PRIVATE -Wall -Wextra -Wpedantic)
add_executable(native_alsa_host service.cpp)
target_compile_definitions(native_alsa_host PRIVATE NATIVE_ALSA_HOST)
target_link_libraries(native_alsa_host PRIVATE native_audio_pump native_alsa Threads::Threads)
target_compile_options(native_alsa_host PRIVATE -Wall -Wextra -Wpedantic)
if(BUILD_TESTING)
  add_executable(native_audio_pump_tests tests/audio_pump_tests.cpp)
  target_link_libraries(native_audio_pump_tests PRIVATE native_audio_pump)
  add_test(NAME native_audio_pump COMMAND native_audio_pump_tests)
  add_executable(native_frame_tests tests/frame_tests.cpp)
  target_link_libraries(native_frame_tests PRIVATE native_audio_pump)
  add_test(NAME native_frame COMMAND native_frame_tests)
  add_executable(native_spsc_queue_tests tests/spsc_queue_tests.cpp)
  target_link_libraries(native_spsc_queue_tests PRIVATE native_audio_pump Threads::Threads)
  add_test(NAME native_spsc_queue COMMAND native_spsc_queue_tests)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME native_alsa_identity_args COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/alsa_identity_args_test.py $<TARGET_FILE:native_alsa_host>)
  add_test(NAME native_fake_host_ipc COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/service_test.py $<TARGET_FILE:native_fake_host>)
endif()
