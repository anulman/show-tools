cmake_minimum_required(VERSION 3.20)
project(show_tools_sampler LANGUAGES CXX)
add_library(sample_engine src/sample_engine.cpp)
target_include_directories(sample_engine PUBLIC include)
target_compile_features(sample_engine PUBLIC cxx_std_17)
target_compile_options(sample_engine PRIVATE -Wall -Wextra -Wpedantic -fno-exceptions)
include(CTest)
if(BUILD_TESTING)
  add_executable(sample_engine_tests tests/native_tests.cpp)
  target_link_libraries(sample_engine_tests PRIVATE sample_engine)
  add_test(NAME sample_engine COMMAND sample_engine_tests)
endif()
