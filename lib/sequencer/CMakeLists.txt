cmake_minimum_required(VERSION 3.20)
project(show_tools_sequencer LANGUAGES CXX)
if(NOT TARGET sample_engine)
  add_subdirectory(../sampler ${CMAKE_CURRENT_BINARY_DIR}/sampler)
endif()
add_library(audio_sequence_engine src/audio_sequence_engine.cpp)
target_include_directories(audio_sequence_engine PUBLIC include)
target_link_libraries(audio_sequence_engine PUBLIC sample_engine)
target_compile_features(audio_sequence_engine PUBLIC cxx_std_17)
target_compile_options(audio_sequence_engine PRIVATE -Wall -Wextra -Wpedantic -fno-exceptions)
include(CTest)
if(BUILD_TESTING)
  add_executable(audio_sequence_engine_tests tests/native_tests.cpp)
  target_link_libraries(audio_sequence_engine_tests PRIVATE audio_sequence_engine)
  add_test(NAME audio_sequence_engine COMMAND audio_sequence_engine_tests)
endif()
