cmake_minimum_required(VERSION 3.16)
project(native_alsa LANGUAGES CXX)
add_library(native_alsa STATIC alsa_pcm.cpp)
target_compile_features(native_alsa PUBLIC cxx_std_17)
target_include_directories(native_alsa PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  find_package(ALSA REQUIRED)
  target_link_libraries(native_alsa PUBLIC ALSA::ALSA)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(native_alsa_test tests/alsa_pcm_test.cpp)
  target_link_libraries(native_alsa_test PRIVATE native_alsa)
  add_test(NAME native_alsa_test COMMAND native_alsa_test)
endif()
if(BUILD_TESTING)
  add_executable(native_alsa_fake_test alsa_pcm.cpp tests/fake_alsa_test.cpp)
  target_compile_features(native_alsa_fake_test PRIVATE cxx_std_17)
  target_compile_definitions(native_alsa_fake_test PRIVATE __linux__)
  target_include_directories(native_alsa_fake_test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} tests/fake)
  add_test(NAME native_alsa_fake_test COMMAND native_alsa_fake_test)
endif()
